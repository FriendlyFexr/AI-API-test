'Video Generation' : 'Text Generation'; log(`Generation mode switched to: ${mode}`, 'info'); }); function log(message, type = 'info') { const timestamp = new Date().toLocaleTimeString(); const entry = document.createElement('div'); entry.className = `log-entry log-${type}`; entry.textContent = `[${timestamp}] [${type.toUpperCase()}] ${message}`; debugLog.appendChild(entry); debugLog.scrollTop = debugLog.scrollHeight; } function clearLogs() { debugLog.innerHTML = ''; log('Logs cleared.', 'info'); } async function sendRequest() { const prompt = promptInput.value.trim(); const isVideo = toggle.checked; if (!prompt) { log('Error: Prompt field is empty.', 'error'); return; } if (!GEMINI_API_KEY || GEMINI_API_KEY === "YOUR_GEMINI_API_KEY" || !OPENAI_API_KEY || OPENAI_API_KEY === "YOUR_OPENAI_API_KEY") { log('Error: Please insert valid API keys into the script.', 'error'); return; } submitBtn.disabled = true; log(`Initiating parallel ${isVideo ? 'video' : 'text'} via OpenAI...`; // Run API requests simultaneously Promise.all([ fetchGemini(prompt, isVideo), fetchOpenAI(prompt, isVideo) ]).finally(() => { submitBtn.disabled = false; }); } // ---------------------------------------------------- // GOOGLE GEMINI API FETCH // ---------------------------------------------------- async function fetchGemini(prompt, isVideo) { try { log('[Gemini API] Request sent...', 'info'); if (!isVideo) { // Text Generation via Gemini 1.5 Flash const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent?key=${GEMINI_API_KEY}`; const response = await fetch(url, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ contents: [{ parts: [{ text: prompt }] }] }) }); const data = await response.json(); if (data.error) throw new Error(data.error.message); const resultText = data.candidates[0].content.parts[0].text; geminiOutput.innerHTML = `

${resultText}

`; log('[Gemini API] Text generated successfully.', 'success'); geminiStatus.textContent = 'Complete'; } else { // Video Generation Polling (Asynchronous Job) log('[Gemini API] Submitting Video Generation task...', 'info'); const startUrl = `https://generativelanguage.googleapis.com/v1beta/models/veo-2.0-generate-001:predictLongRunning?key=${GEMINI_API_KEY}`; const response = await fetch(startUrl, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ prompt: { text: prompt } }) }); const job = await response.json(); if (job.error) throw new Error(job.error.message); log(`[Gemini API] Video task started. ${resultText}

`; log('[OpenAI API] Text generated successfully.', 'success'); openaiStatus.textContent = 'Complete'; } else { // Video Generation via OpenAI Sora API log('[OpenAI API] Requesting video generation...', 'info'); const response = await fetch('https://api.openai.com/v1/videos', { method: 'POST', headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${OPENAI_API_KEY}` }, body: JSON.stringify({ model: 'sora-1.0', prompt: prompt, size: '1280x720' }) }); const data = await response.json(); if (data.error) throw new Error(data.error.message); // Poll video status await pollOpenAIVideo(data.id, openaiOutput, openaiStatus); } } catch (err) { log(`[OpenAI API] Error: ${err.message}`, 'error'); openaiOutput.innerHTML = `Error: ${err.message}`; openaiStatus.textContent = 'Failed'; } } // Polling Helper for Video Operations async function pollVideoJob(operationName, apiKey, outputElement, statusElement, providerName) { const pollUrl = `https://generativelanguage.googleapis.com/v1beta/${operationName}?key=${apiKey}`; let isDone = false; while (!isDone) { await new Promise(r => setTimeout(r, 5000)); // Wait 5s between checks log(`[${providerName} API] Polling video render status...`, 'info'); const res = await fetch(pollUrl); const data = await res.json(); if (data.done) { isDone = true; if (data.response && data.response.generatedVideos) { const videoUri = data.response.generatedVideos[0].videoUri; outputElement.innerHTML = ` `; const data = await res.json(); if (data.status === 'succeeded') { isDone = true; outputElement.innerHTML = ` `; log('[OpenAI API] Video generation completed!', 'success'); statusElement.textContent = 'Complete'; } else if (data.status === 'failed') { throw new Error(data.error || 'Video rendering failed'); } } }
